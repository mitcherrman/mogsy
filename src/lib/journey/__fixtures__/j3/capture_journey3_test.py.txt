Drives REAL Bot matches on a seeded copy of the canonical DB (J3
`journey3/daily`) whose ONLY module is exactly the Daily's Journey module --
built by the Daily recipe's own spec builders from a real `journey_slice`
product (`_journey_spec` for Standard module 10: 5 children, 150 s pooled
ACTIVE time; `_survival_journey_spec` for a Survival slot: 3 children, 30 s
each). Each snapshot is the exact JSON the REAL HTTP route
`GET /api/ranked/matches/{id}` returns (FastAPI TestClient over `api_server.app`),
read at a controlled server instant. NO guard narrowing: the real public guard
runs unmodified. Private answers are written to a SEPARATE file for frontend
tests only (to assert ABSENCE from the DOM before a reveal).
"""

def _get(http, mid, at):
    http["clock"]["now"] = at
    r = http["client"].get(f"/api/ranked/matches/{mid}")
    assert r.status_code == 200, r.text
    return r.json()


def _format(db, fid, recipe_id, plan, *, ruleset=None):
    from content_sets import products as P
    from daily_challenge import recipe as stage_recipe
    from ranked_formats import RankedFormat
    from ranked_formats.schema import SCORING_MODEL_POINTS
    from ranked_public import format_config
    conn = sqlite3.connect(db)
    conn.row_factory = sqlite3.Row
    try:
        n = 5 if plan == "standard" else 3
        product = P.check(conn.cursor(), P.JOURNEY_SLICE, (recipe_id,), n, DAY)
    finally:
        conn.close()
    spec = (stage_recipe._journey_spec(product, 10) if plan == "standard"
            else stage_recipe._survival_journey_spec(product, 6))
    kw = dict(format_id=fid, format_version=1, status="active",
              segment_pattern=(spec,), bot_eligible=True,
              rating_eligible=False, scoring_model=SCORING_MODEL_POINTS,
              match_length=1)
    if ruleset is not None:
        kw["ruleset"] = ruleset
    fmt = RankedFormat(**kw).validate()
    format_config.save_config(db, "public", fmt)


def _payloads(db, mid):
    conn = sqlite3.connect(db)
    conn.row_factory = sqlite3.Row
    try:
        row = conn.execute("SELECT * FROM ranked_rounds WHERE match_id = ? "
                           "AND round_number = 1", (mid,)).fetchone()
        return (json.loads(row["segment_payload_json"]),
                json.loads(row["segment_private_json"]))
    finally:
        conn.close()


def _dump(name, snaps, answers):
    os.makedirs(OUT, exist_ok=True)
    with open(os.path.join(OUT, f"{name}.json"), "w", encoding="utf-8") as fh:
        json.dump(snaps, fh, separators=(",", ":"), ensure_ascii=False, default=str)
    with open(os.path.join(OUT, "answers", f"{name}.answers.json"), "w",
              encoding="utf-8") as fh:
        json.dump(answers, fh, indent=1, ensure_ascii=False, default=str)


def _walk(http, key, plan, *, wrong=(), stop_after=None, ruleset=None,
          think_s=8, timeout_child=None):
    from ranked_public import service
    os.makedirs(os.path.join(OUT, "answers"), exist_ok=True)
    db = http["db"]
    recipe_id = LAUNCH[key]
    mid = f"ui3-{key}-{plan}" + ("-stop" if stop_after is not None else "")
    _format(db, mid, recipe_id, plan, ruleset=ruleset)
    service.create_bot_match(db, match_id=mid, human_user_id=A,
                             human_class="tank", now=T0)
    public, private = _payloads(db, mid)
    count = len(private["challenges"])
    answers = [{"index": i, "correct_answer": c["correct_answer"],
                "answer_options": c.get("answer_options")}
               for i, c in enumerate(private["challenges"])]
    delays = public["journey"]["open_delays_ms"]
    snaps = []

    def snap(label, at):
        snaps.append({"label": label, "at": at.isoformat(),
                      "envelope": _get(http, mid, at)})
        return snaps[-1]["envelope"]

    env = snap("child0-leadin", T0 + timedelta(milliseconds=10))
    opened = datetime.fromisoformat(
        env["payload"]["segment_state"]["own_card_started_at"])
    snap("child0-open", opened + timedelta(milliseconds=100))
    for index in range(count):
        snap(f"child{index}-live", opened + timedelta(seconds=think_s / 4))
        at = opened + timedelta(seconds=think_s)
        if index == timeout_child:
            # Never answered: read after its clock has run out.
            env = snap(f"child{index}-timeout",
                       opened + timedelta(seconds=61))
            break
        row = private["challenges"][index]
        choice = row["correct_answer"]
        if index in wrong:
            choice = next(o for o in row["answer_options"] if o != choice)
        service.submit_challenge(db, mid, A, 1, index, {"selected": choice}, at)
        snap(f"child{index}-reveal", at + timedelta(milliseconds=500))
        snap(f"child{index}-reveal-late", at + timedelta(milliseconds=1500))
        if stop_after is not None and index == stop_after:
            snap(f"child{index}-stopped", at + timedelta(seconds=5))
            break
        if index == count - 1:
            snap("finished", at + timedelta(seconds=5))
            break
        delay = int(delays[index + 1]) if index + 1 < len(delays) else 0
        if delay:
            snap(f"child{index + 1}-beat",
                 at + timedelta(milliseconds=REVEAL_MS + delay // 2))
        env = snap(f"child{index + 1}-open",
                   at + timedelta(milliseconds=REVEAL_MS + delay + 100))
        opened = datetime.fromisoformat(
            env["payload"]["segment_state"]["own_card_started_at"])
    return snaps, answers


@pytest.mark.parametrize("key", list(LAUNCH))
def test_capture_standard(http, key):
    # Child 1 answered WRONG on purpose: the Journey still advances, and its
    # reveal still teaches (J3 learner ledger).
    snaps, answers = _walk(http, key, "standard", wrong=(1,))
    _dump(f"{key}.standard", snaps, answers)


@pytest.mark.parametrize("key", list(LAUNCH))
def test_capture_survival(http, key):
    snaps, answers = _walk(http, key, "survival", wrong=())
    _dump(f"{key}.survival", snaps, answers)


def test_capture_survival_strikeout(http):
    """Strike-out INSIDE a Journey: two strikes allowed; wrong on child 0 and
    child 1 ends the stage on child 1 of 3 (child 2 never reached)."""
    from ranked_rulesets import survival
    snaps, answers = _walk(http, "pantheon", "survival", wrong=(0, 1),
                           stop_after=1, ruleset=survival(max_strikes=2))
    _dump("pantheon.survival.strikeout", snaps, answers)


def test_capture_standard_exhaustion(http):
    """The pooled clock runs out on child 1: the Journey ends there."""
    snaps, answers = _walk(http, "voli", "standard", think_s=100,
                           timeout_child=1)
    _dump("voli.standard.exhaust", snaps, answers)
